\element{coordonnees.1}{
  \begin{question}{volumeSpherique}\bareme{1}
    Le volume élémentaire s'écrit en coordonnées sphériques
    \begin{multicols}{2}
      \begin{reponses}
        \bonne{$dV=r^2\sin\theta drd\theta d\varphi$}
        \mauvaise{$dV=r^2\sin\varphi drd\theta d\varphi$}
        \mauvaise{$dV=dr^2\sin\varphi d\theta d\varphi$}
        \mauvaise{$dV=dr^2\cos\varphi d\theta d\varphi$}
      \end{reponses}
    \end{multicols}
  \end{question}
}
\element{coordonnees.1}{

    \begin{multicols}{2}
      \begin{reponses}
        \bonne{$\vec{dS}=rd\theta dz \vec{u_r}$}
        \mauvaise{$\vec{dS}=drdz \vec{u_z}$}
        \mauvaise{$\vec{dS}=dr\sin \theta \vec{u_\theta}$}
        \mauvaise{$\vec{dS}=drd\theta dz \vec{u_r}$}
      \end{reponses}
    \end{multicols}
  \end{question}
}

% Schémas des vecteurs de base, en perspective cavalière comme ceux du cours.
% Un seul vecteur est tracé depuis M, toujours étiqueté \vv{e_\theta} :
% #1 désigne son extrémité (er, et, -et, ez en cylindriques ;
% er, et, -et, ephi en sphériques), #2 place l'étiquette.
\newcommand{\schemaBaseCylindrique}[2]{%
  \begin{tikzpicture}[x={(-0.45cm,-0.45cm)}, y={(1cm,0cm)}, z={(0cm,1cm)}, >=latex, font=\small, baseline=(current bounding box.center)]
    \pgfmathsetmacro{\cosT}{cos(45)}
    \pgfmathsetmacro{\sinT}{sin(45)}
    \draw[->] (0,0,0) -- (1.4,0,0) node[left] {$x$};
    \draw[->] (0,0,0) -- (0,2.3,0) node[below] {$y$};
    \draw[->] (0,0,0) -- (0,0,2) node[left] {$z$};
    \coordinate (M) at ({1.5*\cosT},{1.5*\sinT},1.1);
    \draw[dashed] (0,0,1.1) -- (M) -- ({1.5*\cosT},{1.5*\sinT},0) -- (0,0,0);
    \draw[dashed] (0,0,0) -- (M);
    % \draw[dotted, domain=0:90, variable=\a, smooth] plot ({1.5*cos(\a)},{1.5*sin(\a)},1.1);
    % \draw[->, domain=0:45, variable=\a, smooth] plot ({0.6*cos(\a)},{0.6*sin(\a)},0);
    % \node at ({1.05*cos(22.5)},{1.05*sin(22.5)},0) {$\theta$};
    \coordinate (er) at ({2.3*\cosT},{2.3*\sinT},1.1);
    \coordinate (et) at ({1.5*\cosT-0.8*\sinT},{1.5*\sinT+0.8*\cosT},1.1);
    \coordinate (-et) at ({1.5*\cosT+0.8*\sinT},{1.5*\sinT-0.8*\cosT},1.1);
    \coordinate (ez) at ({1.5*\cosT},{1.5*\sinT},1.9);
    \draw[->, thick] (M) -- (#1) node[#2] {$\vv{e_\theta}$};
    \fill (M) circle (1.2pt) node[above right] {$M$};
  \end{tikzpicture}%
}
\newcommand{\schemaBaseSpherique}[2]{%

  \begin{question}{eThetaCylindriques}\bareme{1}

      \begin{reponses}
        \bonne{\schemaBaseCylindrique{et}{right}}

        \mauvaise{\schemaBaseCylindrique{ez}{right}}
      \end{reponses}
    \end{multicols}
  \end{question}
